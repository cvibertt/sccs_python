library(SCCS)

apdat <- apdat[, c("case", "sta", "end", "stro", "ap", "endap", "cen", "dem")]

# First, identify unique cases
unique_cases <- unique(apdat$case)
n_unique <- length(unique_cases)

# Generate demographic data for unique cases only
set.seed(123)  # For reproducibility

# Generate data for new columns (one value per unique case)
age_at_first_antipsych <- round(runif(n_unique, min=30, max=80), 1)
weight <- round(rnorm(n_unique, mean=90, sd=10), 1)
height <- round(rnorm(n_unique, mean=160, sd=30), 0)
income_level <- sample(1:3, n_unique, replace=TRUE)

# Generate random ethnicity strings
ethnicity_options <- c("white", "black", "latin american", "asian")
ethnicity <- sample(ethnicity_options, n_unique, replace = TRUE)

# Add five stroke-relevant demographic variables
set.seed(456)  # Different seed for stroke variables

hypertension <- sample(0:1, n_unique, replace = TRUE, prob = c(0.7, 0.3))
diabetes <- sample(0:1, n_unique, replace = TRUE, prob = c(0.8, 0.2))
smoking_status <- sample(0:2, n_unique, replace = TRUE, prob = c(0.6, 0.25, 0.15))
family_hx_stroke <- sample(0:1, n_unique, replace = TRUE, prob = c(0.75, 0.25))
atrial_fib <- sample(0:1, n_unique, replace = TRUE, prob = c(0.85, 0.15))

# Create a data frame with demographics for unique cases
demo_df <- data.frame(
  case = unique_cases,
  age_at_first_antipsych = age_at_first_antipsych,
  weight = weight,
  height = height,
  ethnicity = ethnicity,
  income_level = income_level,
  hypertension = hypertension,
  diabetes = diabetes,
  smoking_status = smoking_status,
  family_hx_stroke = family_hx_stroke,
  atrial_fib = atrial_fib
)

# Merge demographics back to apdat based on case ID
apdat <- merge(apdat, demo_df, by = "case", all.x = TRUE)

# Sort by case to maintain original order (optional)
apdat <- apdat[order(apdat$case), ]

# View the updated dataset
head(apdat, 15)

write.csv(apdat, "apdat.csv", row.names = FALSE)

agedem <- floor(seq(70, 95, 5)*365.25)

ap.mod1 <- standardsccs(event ~ ap + age, 
                        indiv=case, 
                        astart=sta, 
                        aend=end, 
                        aevent=stro,
                        adrug=ap,
                        aedrug=endap,
                        washout=c(1,92,182),
                        agegrp=agedem,
                        data=apdat)
